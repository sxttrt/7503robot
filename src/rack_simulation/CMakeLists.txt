cmake_minimum_required(VERSION 3.16)
project(rack_simulation)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(rack_hybrid_actuator SHARED src/hybrid_actuator.cc)
target_compile_features(rack_hybrid_actuator PRIVATE cxx_std_17)
target_link_libraries(rack_hybrid_actuator PRIVATE gz-sim8::gz-sim8 gz-transport13::core gz-plugin2::register)
install(TARGETS rack_hybrid_actuator LIBRARY DESTINATION lib)
ament_environment_hooks("${CMAKE_CURRENT_SOURCE_DIR}/hooks/rack_simulation.dsv.in")
if(BUILD_TESTING)
  enable_testing()
  add_executable(actuator_core_test test/actuator_core_test.cc)
  target_include_directories(actuator_core_test PRIVATE src)
  target_compile_features(actuator_core_test PRIVATE cxx_std_17)
  add_test(NAME actuator_core_test COMMAND actuator_core_test)
endif()
ament_package()
